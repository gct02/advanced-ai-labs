Exercise 1

a)

-------------------------
i.

Π_1 = (V_1, I_1, O_1, γ_1)

V_1 = {a, b}
I_1 = {a:=T, b:=F}
O_1 = ((⊤, ((a ▷ ¬a) ∧ b), 1),
       (⊤, ((b ▷ ¬b) ∧ a), 1))

γ_1 = (a ∧ b)

-------------------------
ii.

Π_2 = (V_2, I_2, O_2, γ_2)

V_2 = {a, b}
I_2 = {a:=F, b:=F, c:=F, d:=F}
O_2 = ((¬a, (b ∧ a), 1),
       (a, ¬a, 1))

γ_2 = (a ∧ b)

-------------------------
iii.

Π_3 = (V_3, I_3, O_3, γ_3)

V_3 = {a, b, c, d}
I_3 = {a:=T, b:=F}
O_3 = ((⊤, ((a ▷ b) ∧ (¬a ▷ c)), 1),
       (c, d, 1),
       (d, a, 1))

γ_3 = (a ∧ b ∧ c ∧ d)

-------------------------
iv.

Π_n = (V_n, I_n, O_n, γ_n)

V_n = {v_1, ..., v_n}
I_n = {v_i:=T if i≠n else F | i in [1,n]}
O_n = ((⊤, (¬v_1 ∧ ¬v_2 ∧ ... ∧ ¬v_{n-1} ∧ v_n), 1),
       (v_n, v_1, 1),
       ((v_n ∧ v_1), v_2, 1),
       ((v_n ∧ v_1 ∧ v_2), v_3, 1),
       ...,
       ((v_n ∧ v_1 ∧ v_2 ∧ ... ∧ v_{n-2}), v_{n-1}, 1))
       
γ_n = (v_1 ∧ ... ∧ v_n)
